-- 家园 HUD：ModelStore 只读订阅刷新金币/等级/容量；生产入口打开 Bag/Shop。
-- 保留账户登出；不再占用 AccountService.onRestored 单槽。

local UIWindow = require("UI/UIWindow")
local UI = require("UI/UIManager")
local ModelStore = require("Models/ModelStore")
local AccountService = require("AccountService")
local ProgressionPresenter = require("UI/Presenters/ProgressionPresenter")
local ProgressionService = require("Services/ProgressionService")
local Catalog = require("Services/GameplayDisplayCatalog")

local M = {}

function M.new()
  local self = UIWindow.new()
  self._clicks = {}
  self._sub = nil
  self._gen = 0
  self._progression = nil

  local function bump()
    self._gen = self._gen + 1
    return self._gen
  end

  local function refresh(snap)
    if not self.view then return end
    local playerId = "0"
    local nickname = ""
    local level = 0
    local gold = "0"
    local bagSlots = 0
    local bag = {}
    if type(snap) == "table" then
      playerId = tostring(snap.playerId or "0")
      nickname = tostring(snap.nickname or "")
      level = snap.level or 0
      gold = tostring(snap.gold or "0")
      bagSlots = snap.bagSlots or 0
      bag = type(snap.bag) == "table" and snap.bag or {}
    end
    local occupied = Catalog.countOccupied(bag)
    self.view:SetText("PlayerId", "ID: " .. playerId)
    self.view:SetText("Nickname", nickname)
    self.view:SetText("Level", "Lv." .. tostring(level))
    pcall(function() self.view:SetText("Gold", "金币: " .. gold) end)
    pcall(function()
      self.view:SetText("BagSlots", string.format("背包: %s/%s", tostring(occupied), tostring(bagSlots)))
    end)
    if self._progression then
      self._progression:showUnlocks(level)
    end
  end

  function self:OnCreate(view)
    UIWindow.OnCreate(self, view)
    view:SetText("Title", "家园")
    self._progression = ProgressionPresenter.new(ProgressionService, ModelStore, nil, Catalog)
    self._progression:OnCreate(view)

    self._clicks[#self._clicks + 1] = view:SubscribeClick("BagButton", function()
      UI.open("Bag", {})
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("ShopButton", function()
      UI.open("Shop", {})
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("FarmButton", function()
      pcall(function()
        CS.Game.Framework.Home.LuaHomeFarmFacade.EnsureRuntime()
      end)
      UI.open("Farm", {})
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("GatherButton", function()
      pcall(function()
        CS.Game.Framework.Home.LuaHomeFarmFacade.EnsureRuntime()
      end)
      UI.open("Gather", {})
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("ToolButton", function()
      UI.open("Tool", {})
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("LogoutButton", function()
      if self._busy then return end
      self._busy = true
      self._op = AccountService.logout(function(code, _)
        self._busy = false
        -- Logout 已 EnterLogin；窗口由 AccountFlow 按 CurrentAddress 统一打开，禁止抢先 Open。
      end)
    end)
  end

  function self:OnOpen(argsJson)
    UIWindow.OnOpen(self, argsJson)
    local gen = bump()
    if self._sub then
      pcall(function() self._sub.dispose() end)
      self._sub = nil
    end
    self._sub = ModelStore.subscribe(nil, function(snap, _)
      if self._gen ~= gen or not self.view then return end
      refresh(snap)
    end)
    -- Progression 由 HUD 统一订阅驱动 showUnlocks，避免第二路 ModelStore 租约。
    refresh(ModelStore.getSnapshot())
  end

  function self:OnClose()
    bump()
    if self._sub then
      pcall(function() self._sub.dispose() end)
      self._sub = nil
    end
    if self._op then
      pcall(function() self._op:Dispose() end)
      self._op = nil
    end
    UIWindow.OnClose(self)
  end

  function self:OnDestroy()
    bump()
    if self._sub then
      pcall(function() self._sub.dispose() end)
      self._sub = nil
    end
    for _, c in ipairs(self._clicks) do
      pcall(function() c:Dispose() end)
    end
    self._clicks = {}
    if self._progression then
      self._progression:OnDestroy()
      self._progression = nil
    end
    UIWindow.OnDestroy(self)
  end

  return self
end

return M
