-- 生产农场窗：选种/模式/状态倒计时；世界手势经 Home 运行时进入 Presenter。

local UIWindow = require("UI/UIWindow")
local UI = require("UI/UIManager")
local ModelStore = require("Models/ModelStore")
local FarmPresenter = require("UI/Presenters/FarmPresenter")
local FarmService = require("Services/FarmService")
local Toast = require("UI/ToastService")

local M = {}

function M.new()
  local self = UIWindow.new()
  self._clicks = {}
  self._presenter = nil

  function self:OnCreate(view)
    UIWindow.OnCreate(self, view)
    self._presenter = FarmPresenter.new(FarmService, ModelStore, nil, nil)
    self._presenter:OnCreate(view)

    self._clicks[#self._clicks + 1] = view:SubscribeClick("PrevSeedButton", function()
      if self._presenter then self._presenter:selectPrevSeed() end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("NextSeedButton", function()
      if self._presenter then self._presenter:selectNextSeed() end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("PlantModeButton", function()
      if self._presenter then self._presenter:setToolMode("plant") end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("WaterModeButton", function()
      if self._presenter then self._presenter:setToolMode("water") end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("FertilizeModeButton", function()
      if self._presenter then self._presenter:setToolMode("fertilize") end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("HarvestModeButton", function()
      if self._presenter then self._presenter:setToolMode("harvest") end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("CountdownPrevButton", function()
      if self._presenter then self._presenter:countdownPrevPage() end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("CountdownNextButton", function()
      if self._presenter then self._presenter:countdownNextPage() end
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("CloseButton", function()
      UI.close("Farm")
    end)
  end

  function self:OnOpen(argsJson)
    UIWindow.OnOpen(self, argsJson)
    pcall(function()
      CS.Game.Framework.Home.LuaHomeFarmFacade.EnsureRuntime()
    end)
    if self._presenter then self._presenter:OnOpen(argsJson) end
  end

  function self:OnClose()
    if self._presenter then self._presenter:OnClose() end
    UIWindow.OnClose(self)
  end

  function self:OnDestroy()
    for _, c in ipairs(self._clicks) do
      pcall(function() c:Dispose() end)
    end
    self._clicks = {}
    if self._presenter then
      self._presenter:OnDestroy()
      self._presenter = nil
    end
    UIWindow.OnDestroy(self)
  end

  return self
end

return M
