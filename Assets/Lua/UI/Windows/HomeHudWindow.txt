-- 空家园 HUD：显示 playerId / 昵称 / 等级；登出回 Login 并清 Model/Token。

local UIWindow = require("UI/UIWindow")
local PlayerModel = require("PlayerModel")
local AccountService = require("AccountService")

local M = {}

function M.new()
  local self = UIWindow.new()
  self._clicks = {}

  local function refresh()
    if not self.view then return end
    self.view:SetText("PlayerId", "ID: " .. tostring(PlayerModel.playerId))
    self.view:SetText("Nickname", tostring(PlayerModel.nickname))
    self.view:SetText("Level", "Lv." .. tostring(PlayerModel.level))
  end

  function self:OnCreate(view)
    UIWindow.OnCreate(self, view)
    view:SetText("Title", "家园")
    refresh()
    -- 断线自动恢复后快照已覆盖 Model；本窗仍打开时须刷新可见 HUD。
    AccountService.onRestored(function(_, _)
      refresh()
    end)
    self._clicks[#self._clicks + 1] = view:SubscribeClick("LogoutButton", function()
      if self._busy then return end
      self._busy = true
      self._op = AccountService.logout(function(code, _)
        self._busy = false
        -- Logout 已 EnterLogin；窗口由 AccountFlow 按 CurrentAddress 统一打开，禁止抢先 Open。
      end)
    end)
  end

  function self:OnOpen(argsJson)
    UIWindow.OnOpen(self, argsJson)
    refresh()
  end

  function self:OnClose()
    if self._op then
      pcall(function() self._op:Dispose() end)
      self._op = nil
    end
    UIWindow.OnClose(self)
  end

  function self:OnDestroy()
    for _, c in ipairs(self._clicks) do
      pcall(function() c:Dispose() end)
    end
    self._clicks = {}
    UIWindow.OnDestroy(self)
  end

  return self
end

return M
