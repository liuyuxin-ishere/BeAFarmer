-- 兼容只读门面：require("PlayerModel") 保留。唯一写入口为 Models/ModelStore。
-- 不提供公开可变字段写与 applySnapshot 旁路。

local Store = require("Models/ModelStore")

local M = {}

local function read(key, default)
  local v = Store._compat_get(key)
  if v == nil then
    return default
  end
  return v
end

function M.snapshot()
  local s = Store.getSnapshot()
  if s == nil then
    return {
      playerId = "0",
      nickname = "",
      level = 0,
      gold = "0",
      bagCount = 0,
      plotCount = 0,
      hasOrderProgress = false,
    }
  end
  return {
    playerId = s.playerId,
    nickname = s.nickname,
    level = s.level,
    gold = s.gold,
    bagCount = type(s.bag) == "table" and #s.bag or 0,
    plotCount = type(s.plots) == "table" and #s.plots or 0,
    hasOrderProgress = s.orderProgress ~= nil,
  }
end

-- 仅会话拥有者（AccountService / 模型端口）可清；Window 不应调用。
function M.clear()
  Store.clear("player-model-clear")
end

-- 封闭旧旁路：引导调用方迁移到 ModelStore.replaceSnapshot
function M.applySnapshot()
  error("PlayerModel.applySnapshot removed; use Models.ModelStore.replaceSnapshot")
end

setmetatable(M, {
  __index = function(_, key)
    if key == "playerId" then return read("playerId", "0") end
    if key == "nickname" then return read("nickname", "") end
    if key == "level" then return read("level", 0) end
    if key == "exp" then return read("exp", 0) end
    if key == "gold" then return read("gold", "0") end
    if key == "gem" then return read("gem", "0") end
    if key == "ticket" then return read("ticket", "0") end
    if key == "shard" then return read("shard", "0") end
    if key == "bagSlots" then return read("bagSlots", 0) end
    if key == "guideStep" then return read("guideStep", 0) end
    if key == "bag" then return read("bag", {}) end
    if key == "plots" then return read("plots", {}) end
    if key == "nodes" then return read("nodes", {}) end
    if key == "tools" then return read("tools", {}) end
    if key == "facilities" then return read("facilities", {}) end
    if key == "orders" then return read("orders", {}) end
    if key == "orderProgress" then return read("orderProgress", nil) end
    if key == "home" then return read("home", nil) end
    if key == "avatar" then return read("avatar", nil) end
    if key == "gachaStates" then return read("gachaStates", {}) end
    if key == "activities" then return read("activities", {}) end
    return nil
  end,
  __newindex = function(_, key, _)
    error("PlayerModel." .. tostring(key) .. " is read-only; use Models.ModelStore")
  end,
})

return M
